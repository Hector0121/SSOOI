function mayusculas                       # Recibe el texto en $1 y deja el resultado en MAYUS
{                                         
	MAYUS=""                              # Vacía MAYUS
	POS_M=0                               # Posición del carácter que se está mirando (empieza en 0, el primero)
	while test $POS_M -lt ${#1}           # Repite mientras POS_M sea menor que la longitud del texto. la long del texto es #1
	do                                    
		CAR="${1:$POS_M:1}"               # Coge 1 carácter de $1 a partir de la posición POS_M (sesión 4: ${VAR:inicio:longitud})
		case "$CAR" in                    # Mira qué carácter es
			[abcdefghijklmnopqrstuvwxyz] )  # Solo entra aquí si es una letra minúscula de la a a la z
				PREVIO="${MINUSCULAS%%$CAR*}"     # De "abcd...z" borra desde la letra hasta el final: quedan las letras ANTERIORES a ella
				CAR="${ABECEDARIO:${#PREVIO}:1}"  # ${#PREVIO} = cuántas letras hay antes = su posición; coge esa misma posición en "ABCD...Z"
				;;                      
		esac                              # Si no era minúscula CAR se queda igual
		MAYUS="$MAYUS$CAR"                # Añade el carácter (convertido o no) al final del resultado
		POS_M=$(($POS_M+1))               # Pasa a la siguiente posición
	done                                  # vuelve al while
}                                         # El resultado queda en la variable MAYUS

function entero # ahora sigo 
{
	NUM="$1"
	case "$NUM" in
		"" | *[!0123456789]* )
			return 1
			;;
	esac
	while test ${#NUM} -gt 1 -a "${NUM:0:1}" = "0"
	do
		NUM="${NUM:1}"
	done
	test ${#NUM} -le 9
}
