function mayusculas
{
	MAYUS=""
	POS_M=0
	while test $POS_M -lt ${#1}
	do
		CAR="${1:$POS_M:1}"
		for I in ${!MINUSCULAS[*]}
		do
				if test "$CAR" = "${MINUSCULAS[$I]}"			
				then
				CAR=${ABECEDARIO[$I]}
			fi
		done
		MAYUS="$MAYUS$CAR"
		POS_M=$(($POS_M+1))
	done
}

function entero
{
	NUM="$1"
	case "$NUM" in
		"" | *[!0123456789]* )
			return 1
			;;
	esac
	while test ${#NUM} -gt 1 -a "${NUM:0:1}" = "0"
	do
		NUM="${NUM:1}"
	done
	test ${#NUM} -le 9
}

function rango
{
	if entero "${VALOR[$1]}"
	then
		if test $NUM -ge $2 -a $NUM -le $3
		then
			VALOR[$1]=$NUM
			return 0
		fi
	fi
	echo "Error: ${PARAM[$1]} debe ser $4 (valor leido: \"${VALOR[$1]}\")."
	CONFIG_OK=1
	return 1
}



function cargar_config
{
	unset VALOR # Borra el vector VALOR para empezar desde 0
	LINEAS_MAL=0
	while leer_linea
	do
		ENCONTRADO=0
		if test -z "$LINEA" #al usar -z, si la cadena esta vacia es verdadero, sino falso
		then
			ENCONTRADO=1
		fi
		for N in 1 2 3 4 5 6 7 8
		do
			case "$LINEA" in
				"${PARAM[$N]}="* ) #parametro que sea carácter = y despues lo que sea que tenga el valor
					VALOR[$N]="${LINEA#*=}"
					ENCONTRADO=1
					;;
			esac
		done
		if test $ENCONTRADO -eq 0
		then
			echo "ERROR: linea no reconocida en $CFG: \"$LINEA\""
			LINEAS_MAL=1
		fi
	done < "$CFG"
	return $LINEAS_MAL
}

function validar_config
{
	CONFIG_OK=0
	rango 3 50 100 "un numero entero entre 50 y 100"
	rango 4 5 10 "un numero entero entre 5 y 10"
	if test $CONFIG_OK -eq 0
	then
		rango 5 $((${VALOR[4]}+1)) $((${VALOR[3]}-1)) "un numero entero que cumpla PFALLO < PRESPUESTA < PVIDA (${VALOR[4]} < PRESPUESTA < ${VALOR[3]})"
	else
		rango 5 1 999999999 "un numero entero que cumpla PFALLO < PRESPUESTA < PVIDA"
	fi
	rango 1 2 4 "un numero entero entre 2 y 4"
	rango 2 1 999999999 "un numero entero positivo"
	rango 6 1 4 "un numero entero entre 1 y 4"
	rango 7 0 2 "un numero entero entre 0 y 2"
	case "${VALOR[8]}" in
		"" | */ )
			echo "Error: LOG debe ser la ruta de un fichero (valor leido: \"${VALOR[8]}\")."
			CONFIG_OK=1
			;;
		* )
			if test -d "${VALOR[8]}"
			then
				echo "Error: LOG (${VALOR[8]}) es un directorio, no un fichero."
				CONFIG_OK=1
			fi
			;;
	esac
	return $CONFIG_OK
}

function aplicar_config
{
	JUGADORES=${VALOR[1]}
	RONDAS=${VALOR[2]}
	PVIDA=${VALOR[3]}
	PFALLO=${VALOR[4]}
	PRESPUESTA=${VALOR[5]}
	CATEGORIAS=${VALOR[6]}

}

function escribir_config
{
	for N in 1 2 3 4 5 6 7 8
	do
		echo "${PARAM[$N]}=${VALOR[$N]}"
	done
}

function cambiar_parametro
{
	NUMERO=$1
	ANTERIOR="${VALOR[$NUMERO]}"
	echo -n " Nuevo valor para ${PARAM[$NUMERO]} (actual: $ANTERIOR) >> "
	leer
	VALOR[$NUMERO]="$ENTRADA"
	if validar_config
	then
		escribir_config > "$CFG"
		aplicar_config
		echo " ${PARAM[$NUMERO]} se ha cambiado a ${VALOR[$NUMERO]} y se ha guardado en $CFG."
	else
		VALOR[$NUMERO]="$ANTERIOR"
		echo " No se ha modificado ${PARAM[$NUMERO]}: sigue valiendo $ANTERIOR."
	fi
}

function frase_incorrecta
{
	case "$1" in
		"" | " "* | *" " | *"  "* | *[!ABCDEFGHIJKLMNOPQRSTUVWXYZ\ ]* )
			return 0
			;;
	esac
	return 1
}

function cargar_frases
{
	declare -A VISTA #sirve para recordar que frases ya han salido
	unset FCAT FNIV FTXT
	NFRASES=0
	NINVALIDAS=0
	NREPETIDAS=0
	while leer_linea
	do
		if test -n "$LINEA"
		then
			analizar_frase
		fi
	done < "$FRASES"
}


function analizar_frase
{
	NIVEL=0
	CATEG="${LINEA%%:*}" #borra todo despues de los : para quedarse solo con la categoria
	TEXTO="${LINEA#*:}" #se queda con todo despues de los : para quedarse con la frase
	for N in 1 2 3 4
	do
		if test "$CATEG" = "${CATS[$N]}"
		then
			NIVEL=$N
		fi
	done
	if test "$CATEG" = "$LINEA"
	then
		NIVEL=0
	elif frase_incorrecta "$TEXTO"
	then
		NIVEL=0
	fi
	if test $NIVEL -eq 0
	then
		NINVALIDAS=$(($NINVALIDAS+1))
	elif test -n "${VISTA[$TEXTO]}"
	then
		NREPETIDAS=$(($NREPETIDAS+1))
	else
		VISTA[$TEXTO]=1
		NFRASES=$(($NFRASES+1))
		FCAT[$NFRASES]="$CATEG"
		FNIV[$NFRASES]=$NIVEL
		FTXT[$NFRASES]="$TEXTO"
	fi
}


function consultar_frases
{
	cargar_frases
	for N in 1 2 3 4
	do
		ESTADO_CAT="habilitada"
		if test $N -gt $CATEGORIAS
		then
			ESTADO_CAT="no habilitada"
		fi
		echo " --- ${CATS[$N]} ($ESTADO_CAT con CATEGORIAS=$CATEGORIAS) ---"
		CUENTA=0
		for POS in ${!FTXT[*]}
		do
			if test ${FNIV[$POS]} -eq $N
			then
				CUENTA=$(($CUENTA+1))
				echo "   $CUENTA. ${FTXT[$POS]}"
			fi
		done
		if test $CUENTA -eq 0
		then
			echo "   (no hay frases de esta categoría)"
		fi
	done
	echo " Total: $NFRASES frases válidas."
	if test $NINVALIDAS -gt 0
	then
		echo " Aviso: se han ignorado $NINVALIDAS líneas con formato incorrecto."
	fi
	if test $NREPETIDAS -gt 0
	then
		echo " Aviso: se han ignorado $NREPETIDAS frases repetidas."
	fi
}


function asegurar_salto
{
	if test -s "$1"
	then
		ULTIMO=`tail -c 1 "$1"`
		if test -n "$ULTIMO"
		then
			echo 2> /dev/null >> "$1"
		fi
	fi
}


function anadir_frase
{
	echo " Categorías:"
	for N in 1 2 3 4
	do
		echo "   $N) ${CATS[$N]}"
	done
	NIVEL=""
	while test -z "$NIVEL"
	do
		echo -n " Elija la categoría (1-4) >> "
		leer
		case "$ENTRADA" in
			1 | 2 | 3 | 4 )
				NIVEL=$ENTRADA
				;;
			* )
				echo " Categoría no válida."
				;;
		esac
	done
	echo -n " Introduzca la frase >> "
	leer
	mayusculas "$ENTRADA"
	if frase_incorrecta "$MAYUS"
	then
		echo " Error: frase no válida. Solo puede contener letras de la A a la Z"
		echo " (sin acentos, sin Ñ ni caracteres especiales) separadas por un único espacio."
	elif grep ":$MAYUS\$" "$FRASES" > /dev/null
	then
		echo " Error: la frase \"$MAYUS\" ya existe en $FRASES."
	else
		asegurar_salto "$FRASES"
		if echo "${CATS[$NIVEL]}:$MAYUS" 2> /dev/null >> "$FRASES"
		then
			echo " Frase añadida: ${CATS[$NIVEL]}:$MAYUS"
		else
			echo " Error: no se ha podido escribir en $FRASES."
		fi
	fi
}



function opcion_frases
{
	SALIR_F=0
	while test $SALIR_F -eq 0
	do
		echo "==========================================================="
		echo "                          FRASES"
		echo "==========================================================="
		echo " 1) Consultar las frases"
		echo " 2) Añadir una frase"
		echo " 0) Volver al menú principal"
		echo -n " Introduzca una opción >> "
		leer
		case "$ENTRADA" in
			1 )
				consultar_frases
				;;
			2 )
				anadir_frase
				;;
			0 )
				SALIR_F=1
				;;
			* )
				echo " Opcion no valida."
				;;
		esac
	done
}

function llamar_frase()
{
    eval "declare -g -a FRASEV=()";
    eval "declare -g -a FRAS_DESCOMPUESTA=()"
    eval "declare -g -a CATEGORIASV=()"
    local ARCHIVO="config.cfg"
    local LINEA_CAT=$(grep "^CATEGORIA" "$ARCHIVO" )
    local CAT_NUM=${LINEA_CAT#*=}""
    CATEGORIASV=()
    case "$CAT_NUM" in


            1)
                CATEGORIASV+=("REFRAN")
               
                ;;
            2)  CATEGORIASV+=("REFRAN")
                CATEGORIASV+=("PELICULA")
               
                ;;
            3)  CATEGORIASV+=("REFRAN")
                CATEGORIASV+=("PELICULA")
                CATEGORIASV+=("MUSICA")
               
                ;;
            4)  CATEGORIASV+=("REFRAN")
                CATEGORIASV+=("PELICULA")
                CATEGORIASV+=("MUSICA")
                CATEGORIASV+=("DEPORTE")
               
                ;;


    esac
    while true; do
        local CAT_CORRECT=0
        local FRASE_ALET=$(shuf -n 1 frases.dat)
        local CATEGORIA="${FRASE_ALET%:*}"
        local FRASE="${FRASE_ALET#*:}"


        for ((i=0;i<${#CATEGORIASV[@]};i++)); do
            if [[ ${CATEGORIASV[i]} == "$CATEGORIA" ]]; then
                CAT_CORRECT=1
                break
            fi
        done


        if [[ $CAT_CORRECT -eq 1 ]]; then
            CAT_CORRECT=0
            break
        fi
    done
    FRASEV+=("$CATEGORIA")
    FRASEV+=("$FRASE")


local LONGITUD=${#FRASE}
FRASEV+=("$LONGITUD")


for(( i=0;i<LONGITUD;i++ )); do


local LETRA="${FRASE:i:1}"
FRAS_DESCOMPUESTA+=("$LETRA")
done


}

for(( i=0;i<LONGITUD;i++)); do

        FRASEOCULTA+=(" ")


else
    FRASEOCULTA+=("_")
fi


done    
}

function IA_random()
{
    local ARCHIVO="config.cfg"
    local LINEA_ESTR=$(grep "^ESTRATEGIA" "$ARCHIVO")
    local NUM_ESTR=${LINEA_ESTR#*=}
    local LETRA
    case "$NUM_ESTR" in
        0)
            for ((POS=0; POS<26; POS++)); do
                LETRA=$POS
                local USADA=0
                for ((p=0; p<${#LETRAS_ESCRITAS[@]}; p++)); do
                    if [[ "${ABECEDARIO[POS]}" == "${LETRAS_ESCRITAS[p]}" ]]; then
                        USADA=1
                    fi
                done
                if [[ $USADA -eq 0 ]]; then
                    break
                fi
            done
            ;;
        1)
            LETRA=$((RANDOM % 26))
            ;;
        2)
            LETRA=$((RANDOM % 26))
            ;;
        *)
            LETRA=0
            ;;
    esac
    echo $LETRA
}

function dividir
{
	CENT=$(( ($1 * 100 + $2 / 2) / $2 ))
	FRAC=$(($CENT % 100))
	if test $FRAC -lt 10
	then
		FRAC="0$FRAC"
	fi
	DEC="$(($CENT / 100)),$FRAC"
}

function columna
{
	grep "|" "$LOG" | grep -v "^FECHA" | cut -d "|" -f $1
}

function sumar
{
	SUMA=0
	for V in `columna $1`
	do
		SUMA=$(($SUMA+$V))
	done
}

function mas_usados
{
	MAXIMO=0
	for V in $2
	do
		VECES[$V]=`columna $1 | grep "^$V\$" | wc -l`
		if test ${VECES[$V]} -gt $MAXIMO
		then
			MAXIMO=${VECES[$V]}
		fi
	done
	for V in $2
	do
		if test ${VECES[$V]} -eq $MAXIMO
		then
			echo "   $3=$V ($MAXIMO partidas)"
		fi
	done
}

function opcion_estadisticas
{
	echo "==========================================================="
	echo "                       ESTADÍSTICAS"
	echo "==========================================================="
	if test ! -f "$LOG"
	then
		echo " No hay partidas registradas: no existe el fichero $LOG."
		return
	fi
	NPART=`columna 1 | wc -l`
	if test $NPART -eq 0
	then
		echo " No hay partidas registradas en $LOG."
		return
	fi
	echo " Partidas jugadas: $NPART"
	sumar 3
	dividir $SUMA $NPART
	echo " Tiempo medio: $DEC segundos"
	MIN_TPO=-1
	MAX_TPO=-1
	for L in `columna 1-3`
	do
		T="${L##*|}"
		if test $MIN_TPO -lt 0 -o $T -lt $MIN_TPO
		then
			MIN_TPO=$T
			MIN_L="$L"
		fi
		if test $T -gt $MAX_TPO
		then
			MAX_TPO=$T
			MAX_L="$L"
		fi
	done
	MIN_L="${MIN_L%|*}"
	MAX_L="${MAX_L%|*}"
	echo " Partida más corta: ${MIN_L/|/ a las }, $MIN_TPO segundos"
	echo " Partida más larga: ${MAX_L/|/ a las }, $MAX_TPO segundos"
	echo " Rendimiento por jugador:"
	for J in 1 2 3 4
	do
		JUGADAS=`columna $(($J+11)) | grep -v "^-\$" | wc -l`
		VICTORIAS=`columna 11 | grep "^$J\$" | wc -l`
		if test $JUGADAS -eq 0
		then
			echo "   Jugador $J: no ha participado en ninguna partida"
		else
			dividir $(($VICTORIAS*100)) $JUGADAS
			echo "   Jugador $J: $VICTORIAS partidas ganadas de $JUGADAS jugadas ($DEC %)"
		fi
	done
	sumar 5
	TOTAL_RONDAS=$SUMA
	dividir $TOTAL_RONDAS $NPART
	echo " Promedio de rondas por partida: $DEC"
	echo " Estrategia de IA más utilizada:"
	mas_usados 10 "0 1 2" ESTRATEGIA
	echo " Nivel de categorías más utilizado:"
	mas_usados 9 "1 2 3 4" CATEGORIAS
	sumar 16
	dividir $SUMA $NPART
	echo " Promedio de letras utilizadas por partida: $DEC"
	sumar 17
	DESIERTAS=$(($TOTAL_RONDAS-$SUMA))
	dividir $(($DESIERTAS*100)) $TOTAL_RONDAS
	echo " Estadística libre - rondas desiertas: $DESIERTAS de $TOTAL_RONDAS rondas jugadas ($DEC %)"
}
